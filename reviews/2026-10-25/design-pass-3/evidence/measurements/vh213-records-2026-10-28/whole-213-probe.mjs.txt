// VH-213's probe, VH-206's (no-audio-206-probe-after.mjs) with its cells
// changed: the before and after of "a whole video with too little audio",
// walked with the pass's walker modules (imported read-only from WALKER; no
// walker file is added or changed by the probe). For each cell group (cell,
// language, width) it runs the scenario's own setup (S-102, S-20, S-99, S-101,

//   WALKER=<walker dir> APP_URL=http://127.0.0.1:5181/ CDP_PORT=9251 \
//   FIX=<fixtures> OUT_ROOT=<scratch>/out RESULTS=<results dir> \

//     node whole-213-probe.mjs [--cells S-102,S-20,S-99,S-101,S-153,whole-390,whole-400,whole-410,whole-cut]
//       [--langs en-GB,zh-Hans,ms-MY] [--views w1280,w320] [--modes light,dark,forced-light,forced-dark]

// ---------------------------------------------------------------------------
const sha = (buf) => createHash('sha256').update(buf).digest('hex')
const treeSha = (dir) => { const h = createHash('sha256'); const walk = (d) => { for (const n of readdirSync(d).sort()) { const p = join(d, n); if (statSync(p).isDirectory()) walk(p); else h.update(p.slice(dir.length) + '\0' + sha(readFileSync(p)) + '\n') } }; walk(dir); return h.digest('hex').slice(0, 16) }
const sh = (cmd, a) => { try { return execFileSync(cmd, a, { encoding: 'utf8' }).trim() } catch (e) { return 'error: ' + String(e.message).slice(0, 120) } }
const power = () => {
  const batt = sh('pmset', ['-g', 'batt']); const pm = sh('pmset', ['-g']).split('\n').filter((l) => /powermode/i.test(l)).map((l) => l.trim()).join('; ')
  const up = sh('uptime', []); const load = (up.match(/load averages?: ([\d.]+)[ ,]+([\d.]+)[ ,]+([\d.]+)/) || []).slice(1).map(Number)
  return { t: new Date().toISOString(), ac: /AC Power/.test(batt), batt: batt.split('\n').map((l) => l.trim()).join(' | '), powermode: pm, uptime: up, load1: load[0], load: load }
}
const until = (w, expr, label, timeout = 120000) => w.waitFor(`(() => { const M = window.__mx; try { return !!(${expr}); } catch (e) { return false; } })()`, { label, timeout })
const mx = (w, expr) => w.evaluate(`(() => { const M = window.__mx; return ${expr}; })()`)

function ffprobe(file) {
  const j = JSON.parse(execFileSync('ffprobe', ['-v', 'error', '-show_entries', 'stream=index,codec_type,codec_name,duration,start_time,nb_frames,sample_rate,channels,width,height,r_frame_rate:format=duration,nb_streams,format_name', '-of', 'json', file], { encoding: 'utf8' }))
  const streams = (j.streams || []).map((s) => ({ index: s.index, kind: s.codec_type, codec: s.codec_name, durationS: s.duration != null ? Number(s.duration) : null, startS: s.start_time != null ? Number(s.start_time) : null, frames: s.nb_frames != null ? Number(s.nb_frames) : null, ...(s.codec_type === 'audio' ? { sampleRate: Number(s.sample_rate), channels: s.channels } : {}), ...(s.codec_type === 'video' ? { size: `${s.width}x${s.height}`, rate: s.r_frame_rate } : {}) }))
  const kinds = streams.map((s) => s.kind)
  return { format: j.format?.format_name, durationS: j.format?.duration != null ? Number(j.format.duration) : null, count: streams.length, video: kinds.filter((k) => k === 'video').length, audio: kinds.filter((k) => k === 'audio').length, other: kinds.filter((k) => k !== 'video' && k !== 'audio').length, streams }
}

/** Every worker's console, all arguments by value (the walker's watchWorkers keeps only two records). */
function watchWorkerLogs(browser, w) {
  const logs = (w.workerLogs = [])
  browser.on(w.page.sessionId, 'Target.attachedToTarget', ({ sessionId, targetInfo }) => {
    if (targetInfo.type !== 'worker') return
    const url = (targetInfo.url || '').replace(/^.*\//, '').slice(0, 60)
    browser.on(sessionId, 'Runtime.consoleAPICalled', async (e) => {
      const t = Date.now()
      const vals = await Promise.all((e.args || []).map(async (a) => {
        if (a.type !== 'object' || !a.objectId) return a.value ?? a.description ?? null
        try { const r = await browser.send('Runtime.callFunctionOn', { objectId: a.objectId, functionDeclaration: 'function () { try { return JSON.stringify(this) } catch (e) { return String(this) } }', returnByValue: true }, sessionId); return JSON.parse(r.result.value) } catch { return a.description ?? null }
      }))
      logs.push({ t, worker: url, type: e.type, args: vals })
    })
  })
}

async function openWalk(browser, name) {
  const w = new Walk(browser, name)
  await w.open()
  w.console = []
  w.page.on('Runtime.consoleAPICalled', (e) => { if (e.type === 'error' || e.type === 'warning') w.console.push({ t: Date.now(), type: e.type, text: e.args.map((a) => a.value ?? a.description ?? '').join(' ').slice(0, 300) }) })
  w.page.on('Runtime.exceptionThrown', (e) => w.console.push({ t: Date.now(), type: 'exception', text: (e.exceptionDetails.exception?.description ?? e.exceptionDetails.text).slice(0, 300) }))
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: QUAL_INIT })
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: MX_INIT })
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: P8_INIT })
  await watchWorkers(browser, w)
  watchWorkerLogs(browser, w)
  return w
}

/** One read per mode, the page's state asserted with each (lang, mode, view; the scenario's state where it has one); light put back. */
async function readModes(w, expr, { lang, view, state }) {
  const out = {}
  const v = VIEWS[view]
  for (const mode of modes) {
    await setMode(w, mode)
    await sleep(250)
    const reached = await assertReached(w, { lang, mode, viewport: { w: v.width, h: v.height, dpr: v.deviceScaleFactor }, state })
    out[mode] = { reached: { outcome: reached.outcome, problems: reached.problems }, ...(await w.evaluate(expr)) }
  }
  await setMode(w, 'light')
  await sleep(200)
  return out
}

async function createJob(w) {
  if (await mx(w, 'M.ack()')) { await w.click(w.acknowledgeButton()); await sleep(200) }
  await w.click(w.createButton())
}

async function saveAndProbe(w, dir, key) {
  const saveSel = '#process-result > .notification[data-kind="success"] .actions > button.button'
  if (!(await mx(w, `!!M.saveBtn() && M.vis(M.saveBtn())`))) return { saved: false, why: 'no Save on the page' }
  const before = (await w.evaluate('window.__audit.saves.length'))
  await w.click(saveSel)
  await w.waitFor(`window.__audit.saves.length > ${before}`, { label: 'the save picker', timeout: 30000 })
  await w.waitSaved()
  const saveStatus = await w.evaluate(`(document.getElementById('save-status') || {}).textContent.replace(/\\s+/g, ' ').trim()`)
  const got = await w.evaluate(`(async () => { const root = await navigator.storage.getDirectory(); const d = await root.getDirectoryHandle('audit-saves'); const name = window.__audit.saves.at(-1).name; const f = await (await d.getFileHandle(name)).getFile(); const b = new Uint8Array(await f.arrayBuffer()); let s = ''; for (let i = 0; i < b.length; i += 0x8000) s += String.fromCharCode.apply(null, b.subarray(i, i + 0x8000)); return { name, size: b.length, b64: btoa(s) }; })()`)
  const buf = Buffer.from(got.b64, 'base64')
  const file = join(dir, `${key}__${got.name.replace(/[^\w.-]+/g, '_')}`)
  writeFileSync(file, buf)
  return { saved: true, saveStatus, name: got.name, bytes: got.size, sha256: sha(buf).slice(0, 16), file, tracks: ffprobe(file) }
}
