// VH-206 #2: WCAG 1.4.12 (Text Spacing) for the kept-part warning, on the
// candidate. The walker's own modules, imported read-only from WALKER (no
// walker file added or changed):
//   - walk.mjs's Walk and its spacing(on): SPACING_CSS (walk.mjs :120)
//     injected as <style id="__audit-spacing"> (walk.mjs :289-296);
//   - wcag-checks.mjs's WCAG_INIT (window.__wcag, textSpacing() at :63).
//     wcag-checks.mjs runs its own walk when imported and does not export
//     WCAG_INIT, so the String.raw literal is read from the file's text (it
//     has no ${} and no backtick inside, so the text is the string's value);
//   - qualify.mjs's QUAL_INIT, VIEWS, setViewport, setMode, assertReached;
//   - scenarios.mjs's SCENARIOS (S-99's and S-20's setup and state), MX_INIT, ready.
// The sliver N14 runs as no-audio-206-probe-after.mjs runs it: ready, End
// time 0:04.0, Tab, settled, the verdict after the trim, 400 ms.
//
// Per cell (cell, language, width), light mode, once the warning is shown:
// read without the sheet, inject the sheet, call textSpacing(), keep what lies
// within #audio-warnings or overlaps it, read the panel's rendered lines per
// item and the page's sideways scroll, remove the sheet.
//
// v2 (after Codex astra's blind check): the positive control now pulls each
// item's detail up onto its heading's first line (position: relative, a top
// measured per item), so the heading and detail text overlap in every cell;
// a cell is "ok" only when every check passes (state reached with and without
// the sheet, the sheet the walker's own and then removed, the read under the
// load bound, and the control catching clipped, spills and overlaps each in
// the panel), otherwise "not ok" with its reasons. --keys picks single cells.
//
//   WALKER=<walker dir> APP_URL=http://127.0.0.1:5181/ CDP_PORT=9251 FIX=<fixtures> \
//   OUT_ROOT=<scratch>/out-ts RESULT=<file.json> SOURCE=<candidate id> \
//     node textspacing-206-v2.mjs [--cells S-99,sliver-N14,S-20] [--langs ...] [--views w1280,w320]
//       [--keys S-99__zh-Hans__w1280,...]

const PANEL = String.raw`

      return { heading: norm(t && t.textContent), headingLines: tl.length, detailLines: dl.length, linesPerItem: tl.length + dl.length, lines: { heading: tl.map((L) => L.text), detail: dl.map((L) => L.text) } }; });

`
// The walker's textSpacing(), unchanged, then each entry resolved back to the
// elements its path names (W.path, the same function, sliced as textSpacing
// slices it) and kept when one lies within #audio-warnings or overlaps it.
const TS = (withSheet) => `(() => { ${PANEL}
  const W = window.__wcag; const ts = W.textSpacing();
  const all = [...document.querySelectorAll('body *')].filter((e) => A.visible(e));
  const inter = (r) => rb && Math.min(r.right, rb.right) - Math.max(r.left, rb.left) > 0.5 && Math.min(r.bottom, rb.bottom) - Math.max(r.top, rb.top) > 0.5;
  const relOf = (e) => (e === box ? 'self' : box.contains(e) ? 'inside' : e.contains(box) ? 'ancestor' : inter(e.getBoundingClientRect()) ? 'intersects' : null);
  const resolve = (p, len) => { const els = all.filter((e) => W.path(e).slice(0, len) === p); const rels = [...new Set(els.map(relOf).filter(Boolean))]; return { matches: els.length, rels }; };
  const keep = (list, len, fields) => list.map((x) => { const r = {}; for (const f of fields) r[f] = resolve(x[f], len); const rels = [...new Set(fields.flatMap((f) => r[f].rels))]; return { ...x, resolved: r, rels }; }).filter((x) => x.rels.length);
  const panelOnly = (list) => list.filter((x) => x.rels.some((r) => r !== 'ancestor'));
  const clipped = keep(ts.clipped, 90, ['el']), spills = keep(ts.spills, 90, ['el']), overlaps = keep(ts.overlaps, 70, ['a', 'b']);
  return { withSheet: ${withSheet}, page, panel: panelRead, geom,
    pageTotals: { clipped: ts.clipped.length, spills: ts.spills.length, overlaps: ts.overlaps.length, spillsCapped: ts.spills.length >= 30, overlapsCapped: ts.overlaps.length >= 30 },
    kept: { clipped, spills, overlaps },
    inPanel: { clipped: panelOnly(clipped).length, spills: panelOnly(spills).length, overlaps: panelOnly(overlaps).length },
    ancestorsOnly: { clipped: clipped.length - panelOnly(clipped).length, spills: spills.length - panelOnly(spills).length, overlaps: overlaps.length - panelOnly(overlaps).length },
    pageAll: ts };
})()`

const sh = (cmd, a) => { try { return execFileSync(cmd, a, { encoding: 'utf8' }).trim() } catch (e) { return 'error: ' + String(e.message).slice(0, 120) } }
const power = () => {
  const batt = sh('pmset', ['-g', 'batt']); const pm = sh('pmset', ['-g']).split('\n').filter((l) => /powermode/i.test(l)).map((l) => l.trim()).join('; ')
  const up = sh('uptime', []); const load = (up.match(/load averages?: ([\d.]+)[ ,]+([\d.]+)[ ,]+([\d.]+)/) || []).slice(1).map(Number)
  return { t: new Date().toISOString(), ac: /AC Power/.test(batt), batt: batt.split('\n').map((l) => l.trim()).join(' | '), powermode: pm, uptime: up, load1: load[0], load }
}
/** Blocks until the 1-minute load is below LOAD_MAX (10 s polls, LOAD_WAIT_MIN bound); records the wait. */
async function underLoad(label) {
  const t0 = Date.now(); let p = power(); const polls = [p.load1]
  while (p.ac && p.load1 >= LOAD_MAX && Date.now() - t0 < LOAD_WAIT_MIN * 60000) { await sleep(10000); p = power(); polls.push(p.load1) }
  return { label, ...p, waitedS: Math.round((Date.now() - t0) / 1000), polls: polls.length > 1 ? polls : undefined, below: p.load1 < LOAD_MAX }
}
const until = (w, expr, label, timeout = 120000) => w.waitFor(`(() => { const M = window.__mx; try { return !!(${expr}); } catch (e) { return false; } })()`, { label, timeout })
const treeSha = (dir) => { const h = createHash('sha256'); const walk = (d) => { for (const n of readdirSync(d).sort()) { const p = join(d, n); if (statSync(p).isDirectory()) walk(p); else h.update(p.slice(dir.length) + '\0' + sha(readFileSync(p)) + '\n') } }; walk(dir); return h.digest('hex').slice(0, 16) }

async function openWalk(browser, name) {
  const w = new Walk(browser, name)
  await w.open()
  w.console = []
  w.page.on('Runtime.consoleAPICalled', (e) => { if (e.type === 'error' || e.type === 'warning') w.console.push({ t: Date.now(), type: e.type, text: e.args.map((x) => x.value ?? x.description ?? '').join(' ').slice(0, 300) }) })
  w.page.on('Runtime.exceptionThrown', (e) => w.console.push({ t: Date.now(), type: 'exception', text: (e.exceptionDetails.exception?.description ?? e.exceptionDetails.text).slice(0, 300) }))
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: QUAL_INIT })
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: MX_INIT })
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: WCAG_INIT })
  return w
}

const run = {

  chrome: browser.version.Browser, node: process.version, walker: WALKER, walkerSha: treeSha(WALKER), appSrcSha: treeSha(join(APP_ROOT, 'src')),

  filter: 'kept: an entry whose path (W.path, sliced as textSpacing slices it: 90 for clipped/spills, 70 for overlaps) names a visible element that is #audio-warnings itself (self), inside it (inside), contains it (ancestor) or whose box intersects its box (intersects). inPanel counts entries with any relation but ancestor.',

    // The instrument's positive control (not a finding), with the sheet on: a
    // deliberately broken panel, so the same filter must keep a clipped
    // detail (max-height 1em, overflow hidden), a spilling heading (height
    // 4px) and the heading's text on the detail's (each item's detail pulled
