// VH-212 #2's driver, modelled read-only on VH-208's (`$M/vh208-records-2026-10-28/vh208-driver.mjs.txt`): VH-212's cells
// walked by the pass's own runner (matrix.mjs), unchanged, on a plan declared in memory, with a read-only probe that
// wraps each walked scenario's `afterMode` in memory (after the runner's setMode, before the state is asserted and the
// instruments run). Nothing here is written into the walker or the repository.
//
//   APP_URL=http://localhost:5181/ CDP_PORT=9251 FIX=<fixtures> OUT_ROOT=<out> RUN_ROOT=<runs> \
//     node vh212-driver.mjs walk --walker <walker dir> --label before --name a \
//       --scenarios S-135,S-136 [--langs …] [--views …] [--modes …] [--cells S-135:en-GB:w320:light,…]
//
// The probe reads, for every cell: the page's language; whether the heavy-limiting warning is on the card (its item
// found by the heading from the page's own table, `warnings.heavyLimiting.heading`, which VH-212 does not change);
// the detail paragraph's full text and own rectangle (the `p.notification-body` in that item: top, bottom, height),
// its line count by the text's client rects (a cross-check of the `measure` instrument's), its line height; the
// warnings block's rectangle (`warnBox`, as VH-208's `named.warnings`), the item's and the heading's, labelled apart;
// and the old and new clause, counted in the detail, in the card, in `#status` (the live region) and in the body, in
// each of the three languages. Its reads go to <RUN_ROOT>/vh212-detail-<plan>.jsonl, one line a record, keyed by the
// walk's name (scenario, language, view) and the mode the runner set (`w.qualMode`).
import { pathToFileURL, fileURLToPath } from 'node:url'
import { join, resolve } from 'node:path'
import { appendFileSync, mkdirSync } from 'node:fs'
import { execFileSync } from 'node:child_process'

const list = (s) => String(s ?? '').split(',').map((x) => x.trim()).filter(Boolean)
function parse(argv) {
  const out = { _: [] }
  for (let i = 0; i < argv.length; i++) {
    if (argv[i].startsWith('--')) { const k = argv[i].slice(2); const v = argv[i + 1] && !argv[i + 1].startsWith('--') ? argv[++i] : true; out[k] = v }
    else out._.push(argv[i])
  }
  return out
}
const args = parse(process.argv.slice(2))

// The clause before VH-212 and the owner's words after it (Backlog review, 2026-10-27), each in its language.
const CLAUSES = {
  old: { 'en-GB': 'so play the new video before you save it', 'zh-Hans': '所以请在保存之前播放新视频', 'ms-MY': 'sebelum anda menyimpannya' },
  new: { 'en-GB': 'so play the new video before you use it', 'zh-Hans': '所以请在使用之前播放新视频', 'ms-MY': 'sebelum anda menggunakannya' },
}

// Read-only: rectangles in document coordinates; text whole; counts of each clause.
const PROBE = String.raw`(() => {
  const CL = ${JSON.stringify(CLAUSES)};
  const M = window.__mx, CARD = '#process-result > .notification[data-kind="success"]';
  const c = document.querySelector(CARD);
  const r = (e) => { if (!e) return null; const b = e.getBoundingClientRect(); return { top: Math.round(b.top + scrollY), bottom: Math.round(b.bottom + scrollY), h: Math.round(b.height), w: Math.round(b.width), hExact: Math.round(b.height * 100) / 100 }; };
  const lang = document.documentElement.lang;
  const heading = M && M.msg ? M.msg('warnings.heavyLimiting.heading') : null;
  const tableDetail = M && M.msg ? M.msg('warnings.heavyLimiting.detail') : null;
  const wb = window.__p8 && window.__p8.warnBox ? window.__p8.warnBox() : null;
  const items = wb ? [...wb.querySelectorAll('li')] : [];
  const li = items.find((el) => { const t = el.querySelector('.notification-item-title'); return !!t && !!heading && t.textContent.trim() === heading; }) || null;
  const ht = li ? li.querySelector('.notification-item-title') : null;
  const p = li ? li.querySelector('p.notification-body') : null;
  let probeLines = null;
  if (p) { const rg = document.createRange(); rg.selectNodeContents(p); probeLines = new Set([...rg.getClientRects()].filter((x) => x.width > 0).map((x) => Math.round(x.top))).size; }
  const n = (s, sub) => (s && sub ? s.split(sub).length - 1 : 0);
  const counts = (s) => { const o = {}; for (const k of ['old', 'new']) { o[k] = {}; for (const [l, sub] of Object.entries(CL[k])) o[k][l] = n(s, sub); } return o; };
  const detail = p ? p.textContent : null;
  const st = document.getElementById('status');
  const cs = p ? getComputedStyle(p) : null;
  return {
    lang, viewport: { w: innerWidth, h: innerHeight },
    card: !!c, cardWhere: c && wb ? (c.contains(wb) ? 'in the card' : 'beside the card') : null,
    heavyPresent: !!p, heading,
    detail, detailIsTable: detail !== null && detail === tableDetail,
    tableClauses: counts(tableDetail),
    detailClauses: counts(detail),
    cardClauses: counts(c ? c.textContent : ''),
    statusClauses: counts(st ? st.textContent : ''),
    bodyClauses: counts(document.body ? document.body.textContent : ''),
    rects: { warnBox: r(wb), item: r(li), heading: r(ht), detailP: r(p) },
    probeLines, lineHeight: cs ? cs.lineHeight : null, fontSize: cs ? cs.fontSize : null,
    itemsInBox: items.length,
  };
})()`

function power() {
  const sh = (cmd, a) => execFileSync(cmd, a, { encoding: 'utf8' })
  const batt = sh('pmset', ['-g', 'batt']), pm = sh('pmset', ['-g']), up = sh('uptime', []).trim()
  const ac = /AC Power/.test(batt), level = Number((batt.match(/(\d+)%/) ?? [])[1])
  const mode = Number((pm.match(/^\s*powermode\s+(\d+)/m) ?? [])[1]), low = /lowpowermode\s+1/.test(pm) || mode === 1
  const load = Number((up.match(/load averages?: ([\d.]+)/) ?? [])[1])
  return { at: new Date().toISOString(), source: ac ? 'AC' : 'battery', level, powermode: mode, low, load, uptime: up, ok: ac || (level > 50 && !low) }
}

async function walk() {
  for (const k of ['APP_URL', 'CDP_PORT', 'FIX', 'OUT_ROOT', 'RUN_ROOT']) if (!process.env[k]) throw new Error('set ' + k)
  if (!args.walker) throw new Error('--walker <walker dir>')
  const walker = resolve(args.walker)
  const label = args.label ?? 'before'
  const scenarios = list(args.scenarios)
  const langs = list(args.langs ?? 'en-GB,zh-Hans,ms-MY'), views = list(args.views ?? 'w1280,w320'), modes = list(args.modes ?? 'light,dark,forced-light,forced-dark')
  const cells = list(args.cells).map((r) => { const [id, lang, view, mode = 'light'] = r.split(':'); return { id, lang, view, mode } })
  const records = [...scenarios.flatMap((id) => langs.flatMap((lang) => views.flatMap((view) => modes.map((mode) => ({ id, lang, view, mode }))))), ...cells]
  if (!records.length) throw new Error('nothing declared')
  const plan = `vh212-${label}-${args.name ?? 'run'}`
  const p = power()
  mkdirSync(process.env.RUN_ROOT, { recursive: true })
  appendFileSync(join(process.env.RUN_ROOT, 'vh212-ledger.jsonl'), JSON.stringify({ plan, label, power: p, walker, appUrl: process.env.APP_URL, cdpPort: process.env.CDP_PORT, fixtures: process.env.FIX, records: records.length, ids: [...new Set(records.map((r) => r.id))] }) + '\n')
  console.log('power: ' + JSON.stringify(p))
  if (!p.ok) { console.log('POWER-STOP: battery at or below 50%, or Low Power'); process.exit(3) }
  const matrixUrl = pathToFileURL(join(walker, 'matrix.mjs')).href
  const { PLANS, SCENARIOS } = await import(new URL('./scenarios.mjs', matrixUrl).href)
  for (const r of records) if (!SCENARIOS[r.id]) throw new Error('the walker holds no scenario ' + r.id)
  const OUT = join(process.env.RUN_ROOT, `vh212-detail-${plan}.jsonl`)
  for (const id of new Set(records.map((r) => r.id))) {
    const sc = SCENARIOS[id], orig = sc.afterMode
    sc.afterMode = async (w) => {
      if (orig) await orig(w)
      let g
      try { g = await w.evaluate(PROBE) } catch (e) { g = { error: String(e.message ?? e).slice(0, 200) } }
      appendFileSync(OUT, JSON.stringify({ plan, walk: w.name, id, mode: w.qualMode, at: new Date().toISOString(), ...g }) + '\n')
    }
  }
  PLANS[plan] = () => records.map((r) => ({ ...r }))
  // matrix.mjs runs its main when it is the entry point: it is made so, unchanged.
  process.argv = [process.argv[0], fileURLToPath(matrixUrl), plan]
  await import(matrixUrl)
}

if (args._[0] === 'walk') await walk()
else { console.log('usage: node vh212-driver.mjs walk --walker <dir> --label <l> --name <n> --scenarios … [--cells …]'); process.exit(2) }
