      expectRow('an error before the job deadline (the declared 900 s): the walk\'s', await judge(w, 'S-140n'), o, { outcome: 'cannot-determine', accepted: null, hang: 'the walk', checks: [['created', (g, ob) => ob.created === true], ['failed in the job', (g, ob) => ob.walkError?.phase === 'job'], ['by that error', (g, ob) => /^a stated error in the walk/.test(ob.walkError?.error ?? '')], ['no deadline reached', (g, ob) => !ob.expired], ['before the deadline', (g, ob) => ob.tEnd < ob.deadlines.job]] })
      await closeW(w)
    }
    // VH-214 #9's cases.
    for (const [name, afterMs] of [['at', 0], ['after', 200]]) {
      // Must 2, an error in the job exactly at the job's deadline (3 s), and one 200 ms after it: F11's silent job runs; the
      // walk's wait for an accepted outcome fails by a stated error (a timer 200 ms before the deadline, then a busy loop
      // to the deadline, or past it): the expiry, judged by the log: no accepted outcome by the deadline, the page's.
      const w = await open(`job-error-${name}-deadline`)
      await applyFault(w, 'F11')
      const realEval = w.evaluate.bind(w)
      const err = { at: null, deadline: null }
      w.evaluate = (expr) => {
        const at = w.obs?.deadlines?.job
        if (!at || err.deadline !== null || !UNTIL_OUTCOME(expr)) return realEval(expr)
        err.deadline = at
        return new Promise((_, rej) => { setTimeout(() => { while (Date.now() < at + afterMs) { /* to the deadline, or past it */ } err.at = Date.now(); rej(new Error(`a stated error in the job, ${afterMs ? `${afterMs} ms after` : 'at'} its deadline`)) }, Math.max(0, at - 200 - Date.now())) })
      }
      const o = await setup(w, 'S-140n', 'en-GB', { jobMs: 3000 })
      w.evaluate = realEval
      out[`jobError_${name}`] = err
      expectRow(`an error in the job ${afterMs ? `${afterMs} ms after` : 'exactly at'} its deadline (3 s), F11's job running: the expiry, the page's`, await judge(w, 'S-140n'), o, { outcome: 'fail', accepted: null, hang: 'the page', checks: [['created', (g, ob) => ob.created === true], [`the error ${afterMs} to ${afterMs + 50} ms from the deadline`, () => err.at !== null && err.at - err.deadline >= afterMs && err.at - err.deadline <= afterMs + 50], ['expired in the job', phase('job')], ['by that error, sent through the expiry', (g, ob) => /^an error at or after it: a stated error in the job/.test(ob.expired?.by ?? '')], ['not the walk\'s error', (g, ob) => !ob.walkError], ['the page answered at expiry with no outcome', (g, ob) => ob.atExpiry?.answered === true && ob.atExpiry.accepted === null]] })
      await releaseAll(w).catch(() => {})
      await closeW(w)
    }
  }
