const sha = (buf) => createHash('sha256').update(buf).digest('hex')
const fileSha = (p) => { const h = createHash('sha256'); h.update(readFileSync(p)); return h.digest('hex').slice(0, 16) }
const treeSha = (dir) => { const h = createHash('sha256'); const walk = (d) => { for (const n of readdirSync(d).sort()) { const p = join(d, n); if (statSync(p).isDirectory()) walk(p); else h.update(p.slice(dir.length) + '\0' + sha(readFileSync(p)) + '\n') } }; walk(dir); return h.digest('hex').slice(0, 16) }
const sh = (cmd, a) => { try { return execFileSync(cmd, a, { encoding: 'utf8' }).trim() } catch (e) { return 'error: ' + String(e.message).slice(0, 120) } }
const power = () => {
  const batt = sh('pmset', ['-g', 'batt']); const pm = sh('pmset', ['-g']).split('\n').filter((l) => /powermode/i.test(l)).map((l) => l.trim()).join('; ')
  const up = sh('uptime', []); const load = (up.match(/load averages?: ([\d.]+)[ ,]+([\d.]+)[ ,]+([\d.]+)/) || []).slice(1).map(Number)
  return { t: new Date().toISOString(), ac: /AC Power/.test(batt), powermode: pm, load1: load[0], load }
}
const byValue = async (browser, args, sessionId) => Promise.all((args || []).map(async (a) => {
  if (a.type !== 'object' || !a.objectId) return a.value ?? a.description ?? null
  try { const r = await browser.send('Runtime.callFunctionOn', { objectId: a.objectId, functionDeclaration: 'function () { try { return JSON.stringify(this) } catch (e) { return String(this) } }', returnByValue: true }, sessionId); return JSON.parse(r.result.value) } catch { return a.description ?? null }
}))

/** Every worker's console, by value; the page's own console, by value; every auto-attached target, passively. */
function watchLogs(browser, w) {
  const workerLogs = (w.workerLogs = []), pageLogs = (w.pageLogs = []), targets = (w.targetsSeen = [])
  browser.on(w.page.sessionId, 'Target.attachedToTarget', ({ sessionId, targetInfo, waitingForDebugger }) => {
    targets.push({ t: Date.now(), type: targetInfo.type, url: (targetInfo.url || '').replace(/^.*\//, '').slice(0, 60), waitingForDebugger, sessionId })
    if (targetInfo.type !== 'worker') return
    const url = (targetInfo.url || '').replace(/^.*\//, '').slice(0, 60)
    browser.on(sessionId, 'Runtime.consoleAPICalled', async (e) => { const t = Date.now(); workerLogs.push({ t, worker: url, type: e.type, args: await byValue(browser, e.args, sessionId) }) })
    browser.on(sessionId, 'Runtime.exceptionThrown', (e) => workerLogs.push({ t: Date.now(), worker: url, type: 'exception', args: [(e.exceptionDetails.exception?.description ?? e.exceptionDetails.text).slice(0, 300)] }))
  })
  browser.on(w.page.sessionId, 'Target.detachedFromTarget', ({ sessionId }) => targets.push({ t: Date.now(), detached: sessionId }))
  w.page.on('Runtime.consoleAPICalled', async (e) => { const t = Date.now(); pageLogs.push({ t, type: e.type, args: await byValue(browser, e.args, w.page.sessionId) }) })
  w.page.on('Runtime.exceptionThrown', (e) => pageLogs.push({ t: Date.now(), type: 'exception', args: [(e.exceptionDetails.exception?.description ?? e.exceptionDetails.text).slice(0, 300)] }))
}

/**
 * heavy-193.mjs's worker watcher, verbatim but for its counter's home
 * (reviews/2026-10-22/design-pass-2/evidence/measurements/walker/heavy-193.mjs:58-73):
 * every auto-attached target waits for the debugger, and only a `worker` is resumed.
 */
async function watchWorkers193(browser, w) {
  const seen = { planned: null, workers: 0 }
  browser.on(w.page.sessionId, 'Target.attachedToTarget', async ({ sessionId, targetInfo }) => {
    if (targetInfo.type !== 'worker') return
    seen.workers++
    browser.on(sessionId, 'Runtime.consoleAPICalled', async (e) => {
      if (e.args[1]?.value !== 'chain planned' || !e.args[2]?.objectId) return
      const r = await browser.send('Runtime.callFunctionOn', { objectId: e.args[2].objectId, functionDeclaration: 'function () { return JSON.stringify(this) }', returnByValue: true }, sessionId)
      seen.planned = JSON.parse(r.result.value)
    })
    await browser.send('Runtime.enable', {}, sessionId).catch(() => {})
    await browser.send('Runtime.runIfWaitingForDebugger', {}, sessionId).catch(() => {})
  })
  await w.page.send('Target.setAutoAttach', { autoAttach: true, waitForDebuggerOnStart: true, flatten: true })
  return seen
}

async function openWalk(browser, name, { replay = false } = {}) {
  const w = new Walk(browser, name)
  await w.open()
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: QUAL_INIT })
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: MX_INIT })
  if (!replay) await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: P8_INIT })
  await w.page.send('Page.addScriptToEvaluateOnNewDocument', { source: VH214_INIT })
  watchLogs(browser, w)
  w.seen193 = replay ? await watchWorkers193(browser, w) : null
  if (!replay) await watchWorkers(browser, w)
  return w
}

/** One read per mode, the page's state asserted with each; light put back. */
async function readModes(w, { lang, view, state }) {
  const out = {}
  const v = VIEWS[view]
  for (const mode of modes) {
    await setMode(w, mode)
    await sleep(250)
    const reached = state ? await assertReached(w, { lang, mode, viewport: { w: v.width, h: v.height, dpr: v.deviceScaleFactor }, state }) : null
    out[mode] = { reached: reached ? { outcome: reached.outcome, problems: reached.problems } : null, ...(await w.evaluate(READ_OUT)) }
  }
  await setMode(w, 'light')
  await sleep(200)
  return out
}

/** A read bounded in time, so a page that no longer answers is recorded as such. */
const bounded = (p, ms, what) => Promise.race([p, new Promise((_, rej) => setTimeout(() => rej(new Error(`bound: ${what} did not return in ${ms / 1000} s`)), ms))])

/** heavy-193.mjs's `bite` routine (its :139-173), step by step and timed, under an outer bound. */
async function replay193(w, file, rec, t0) {
  const steps = (rec.steps = [])
  const step = async (what, fn) => { const s = { what, at: Date.now() - t0 }; steps.push(s); try { s.value = await fn(); s.ms = Date.now() - t0 - s.at; return s.value } catch (e) { s.error = String(e.message ?? e).slice(0, 200); s.ms = Date.now() - t0 - s.at; throw e } }
  const mx = (expr) => w.evaluate(`(() => { const M = window.__mx; return ${expr}; })()`)
  await step('setViewport w1280', () => setViewport(w, 'w1280'))
  await step('setMode light', () => setMode(w, 'light'))
  await step('fresh en-GB', () => fresh(w, 'en-GB'))
  await step('setFiles', () => w.setFiles(file))
  await step('waitSettled (900 s)', () => w.waitSettled({ timeout: 900000 }))
  await step('a verdict (900 s)', () => w.waitFor(`(() => { const M = window.__mx; return M.verdict() || M.kindIn("#source-block") || M.kindIn("#source-report") === "error" })()`, { label: 'a verdict', timeout: 900000 }))
  const canCreate = await step('canCreate', () => w.evaluate(`(() => { const b = document.querySelector(${JSON.stringify(w.createButton())}); return !!b && !b.hidden && !b.disabled })()`))
  if (!canCreate) {
    const reason = await step('reason', () => w.evaluate(`(() => { const t = (s) => { const e = document.querySelector(s); return e && !e.hidden ? e.textContent.replace(/\\s+/g, ' ').trim() : '' }; return [t('#source-block'), t('#source-report'), t('#preflight-report')].filter(Boolean).join(' / ').slice(0, 300) })()`))
    rec.replayOutcome = { created: false, reason }
    return
  }
  await step('create', async () => { if (await mx('M.ack()')) { await w.click(w.acknowledgeButton()); await sleep(200) } await w.click(w.createButton()) })
  await step('"chain planned" or the job ended (1800 s)', async () => {
    const s0 = Date.now()
    while (!w.seen193.planned) {
      if (Date.now() - s0 > 1800000) throw new Error('no "chain planned" in 1800 s')
      const failed = await w.evaluate(`(() => { const p = document.getElementById('process-progress'); const r = document.getElementById('process-result'); return (!p || p.hidden) && !!r && r.children.length > 0 })()`)
      if (failed) throw new Error('the job ended before planning')
      await sleep(500)
    }
  })
  rec.replayOutcome = { created: true, planned: w.seen193.planned }
}
