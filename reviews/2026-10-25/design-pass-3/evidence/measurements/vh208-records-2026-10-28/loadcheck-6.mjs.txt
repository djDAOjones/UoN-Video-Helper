// VH-208 #6: the 1-minute load at each cell's probe time, as loadmap-5.mjs reads it, but over every cell any given run walked:
// prints each cell with no read below load 8 in the runs given (loadmap-5.mjs flagged only cells present in every run given).
// Usage: node loadcheck-6.mjs <run> [<run> …]
import fs from 'node:fs'; import path from 'node:path'
const runs = process.argv.slice(2); const at = {}
for (const run of runs) {
  const L = fs.readFileSync(`${run}.load.log`, 'utf8').split('\n').filter(Boolean).map((l) => { const [t, , v] = l.split(' '); return [t, Number(v)] })
  const day = fs.readFileSync(path.join(run, 'conditions.txt'), 'utf8').match(/start (\d{4}-\d\d-\d\d)/)[1]
  const S = L.map(([t, v]) => [Date.parse(`${day}T${t}Z`), v])
  for (const f of fs.readdirSync(path.join(run, 'runs')).filter((x) => /^vh208-geometry-.*\.jsonl$/.test(x)))
    for (const l of fs.readFileSync(path.join(run, 'runs', f), 'utf8').split('\n').filter(Boolean)) {
      const g = JSON.parse(l); const k = `${g.id}__${g.lang}__${g.walk.split('-').pop()}__${g.mode}`; const t = Date.parse(g.at)
      const prev = S.filter(([s]) => s <= t).pop() ?? S[0]; (at[k] ||= {})[run] = prev[1] } }
const high = Object.entries(at).filter(([, v]) => Object.values(v).every((l) => l >= 8))
console.log(`${Object.keys(at).length} cells in ${runs.join(', ')}; ${high.length} with no read below load 8:`)
for (const [k, v] of high.sort()) console.log(' ', k, JSON.stringify(v))
