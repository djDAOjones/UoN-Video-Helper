#!/bin/zsh
# VH-208 #5: wait (up to 45 min, checking every 60 s) for AC at powermode 2, a 1-minute load below 8 and a free walk lock; then run walk-5.sh once,
# logging load, source, level and mode every 30 s into <name>.load.log and the launcher's output into <name>.launcher.log.
# Usage: zsh go-5.sh <name> <what for the lock> <walker dir> <app root> "<driver args>" …   (the arguments walk-5.sh takes)
D="/Users/joe/Library/Application Support/Claude/side-session-notes/local_e00d677a-46d0-495b-bc55-8ebc86886885/vh208-walk"
NAME=$1
LOCK=/Users/joe/UoN-Video-Helper-side/walk.lock
for f in "$D/$NAME" "$D/$NAME.launcher.log" "$D/$NAME.load.log"; do [ -e "$f" ] && { echo "EXISTS: $f"; exit 2; }; done
s=$(date +%s)
while :; do
  while :; do
    L=$(uptime | sed -E 's/.*load averages?: ([0-9.]+).*/\1/')
    AC=0; pmset -g batt | grep -q 'AC Power' && AC=1
    M=$(pmset -g | sed -nE 's/^ *powermode +([0-9]+).*/\1/p')
    if [ $AC = 1 ] && [ "$M" = 2 ] && [ "$(echo "$L < 8" | bc)" = 1 ] && [ ! -e $LOCK ]; then break; fi
    [ $(( $(date +%s) - s )) -ge 2700 ] && { echo "NOT READY $(date -u +%T) load $L ac $AC powermode $M lock $(cat $LOCK/owner.txt 2>/dev/null | cut -c1-80)"; exit 9; }
    sleep 60
  done
  echo "GO $(date -u +%T) load $L AC powermode $M" >> "$D/$NAME.launcher.log"
  ( while :; do echo "$(date -u +%T) load $(uptime | sed -E 's/.*load averages?: ([0-9.]+).*/\1/') $(pmset -g batt | sed -nE "1s/Now drawing from '(.*)'/\1/p") $(pmset -g batt | grep -oE '[0-9]+%' | head -1) pm$(pmset -g | sed -nE 's/^ *powermode +([0-9]+).*/\1/p')" >> "$D/$NAME.load.log"; sleep 30; done ) &
  LP=$!
  zsh "$D/walk-5.sh" "$@" >> "$D/$NAME.launcher.log" 2>&1
  rc=$?
  kill $LP 2>/dev/null; wait $LP 2>/dev/null
  echo "walk-5 exit $rc $(date -u +%T)" >> "$D/$NAME.launcher.log"
  # 3 (lock taken) or 8 (load rose) between the check and the start means nothing started: wait again
  if { [ $rc = 3 ] || [ $rc = 8 ]; } && [ ! -e "$D/$NAME" ]; then continue; fi
  echo "walk-5 exit $rc"; exit $rc
done
