// VH-229: the readings the table's notes name as the readings, per cell: each later re-walk replaces the
// earlier group's cell, and each error row of the states probe gives way to its re-walk. Reads the records
// only; never writes them. node reconcile.mjs <records dir> > <page>
import { readFileSync, readdirSync } from 'node:fs'
import { join } from 'node:path'
const RD = process.argv[2]
const jl = (f) => readFileSync(join(RD, f), 'utf8').split('\n').filter(Boolean).map((l) => JSON.parse(l))
const key = (r) => [r.id, r.lang, r.view, r.mode].join('__')
const side = (s) => {
  const m = new Map(); let dup = 0, dupDiff = 0
  for (const g of ['a', 'b', 'c', 'c2', 'c3']) for (const r of jl(`${s}-${g}.outcomes.jsonl`)) {
    const k = key(r), old = m.get(k)
    if (old) { dup++; if (!['c2', 'c3'].includes(g) && JSON.stringify([old.reached, old.outcomes, old.rows]) !== JSON.stringify([r.reached, r.outcomes, r.rows])) dupDiff++ }
    m.set(k, { ...r, group: g })
  }
  return { m, dup, dupDiff }
}
const A = side('before'), B = side('after')
const reached = (m) => [...m.values()].filter((r) => r.status === 'ok').length
const moves = {}, alike = {}, add = (o, k, c) => (o[k] ??= []).push(c)
let same = 0, onlyA = [], onlyB = [], reachedMoved = []
for (const [k, a] of A.m) {
  const b = B.m.get(k); if (!b) { onlyA.push(k); continue }
  let moved = false
  if (a.status !== b.status || a.reached !== b.reached) { reachedMoved.push(k); add(moves, `reached: ${a.status}/${a.reached} → ${b.status}/${b.reached}`, k); moved = true }
  for (const n of new Set([...Object.keys(a.outcomes ?? {}), ...Object.keys(b.outcomes ?? {})])) {
    const x = a.outcomes?.[n], y = b.outcomes?.[n]
    if (x !== y) { add(moves, `${n}: ${x} → ${y}`, k); moved = true } else if (x === 'fail') add(alike, n, k)
  }
  const rows = (c) => Object.fromEntries((c.rows ?? []).map((r) => [r.what, r.outcome]))
  const ra = rows(a), rb = rows(b)
  for (const n of new Set([...Object.keys(ra), ...Object.keys(rb)])) {
    if (ra[n] !== rb[n]) { add(moves, `row "${n.slice(0, 60)}": ${ra[n]} → ${rb[n]}`, k); moved = true } else if (ra[n] === 'fail') add(alike, `row "${n.slice(0, 60)}"`, k)
  }
  if (!moved) same++
}
for (const k of B.m.keys()) if (!A.m.has(k)) onlyB.push(k)
const newlyFails = Object.keys(moves).filter((k) => / → fail$/.test(k))
const by = (v) => { const o = {}; for (const c of v) o[c.split('__')[0]] = (o[c.split('__')[0]] ?? 0) + 1; return Object.entries(o).map(([s, n]) => `${s} ${n}`).join(', ') }
const notReached = (s) => { const o = {}; for (const [k, r] of s.m) if (r.status !== 'ok') { const [id, , view, mode] = k.split('__'); const t = `${id} ${view} ${mode}`; o[t] = (o[t] ?? 0) + 1 } return o }
const NA = notReached(A), NB = notReached(B)

// States.
const states = (s) => {
  const m = new Map(); let raw = 0, err = 0
  for (const g of ['states', 'states2']) for (const r of jl(`${s}-${g}.vh229-states.jsonl`)) {
    raw++; const k = [r.lang, r.view, r.mode, r.which, r.state].join('/')
    if (r.error || r.kind === 'error') { err++; continue }
    m.set(k, r)
  }
  return { m, raw, err }
}
const SA = states('before'), SB = states('after')
const flat = (o, pre = '', out = {}) => { for (const [k, v] of Object.entries(o ?? {})) { if (v && typeof v === 'object' && !Array.isArray(v)) flat(v, `${pre}${k}.`, out); else out[`${pre}${k}`] = JSON.stringify(v) } return out }
let both = 0, stSame = 0; const stMoves = {}
for (const [k, a] of SA.m) { const b = SB.m.get(k); if (!b) continue; both++; const fa = flat(a), fb = flat(b); let mv = false; for (const f of new Set([...Object.keys(fa), ...Object.keys(fb)])) if (fa[f] !== fb[f]) { mv = true; stMoves[f] = (stMoves[f] ?? 0) + 1 } if (!mv) stSame++ }
const whichOf = (m) => { const o = {}; for (const r of m.values()) o[r.which] = (o[r.which] ?? 0) + 1; return o }
// Provenance of the before's re-walks.
const prov = readdirSync(RD).filter((f) => /^(before|after)-(c2|c3|states2?|a|b|c)\.conditions\.txt$/.test(f)).map((f) => { const t = readFileSync(join(RD, f), 'utf8'); return [f.replace('.conditions.txt', ''), /diff hash (\w+)/.exec(t)?.[1]] })
const mani = readdirSync(RD).filter((f) => /^before-(c2|c3)\..*manifest\.json$/.test(f)).map((f) => { const j = JSON.parse(readFileSync(join(RD, f), 'utf8')); return `${f.replace('.manifest.json', '')}: dirty ${j.source.dirty}, ${j.source.diffSha256}` })
const probe = readdirSync(RD).filter((f) => /^before-(c2|c3)\.vh229-probe/.test(f)).flatMap((f) => jl(f))
